# 5. Verify

================================================================================
ALTERNATIVE - Manual Fix (if script fails)
================================================================================

# Copy .env to backend
cp .env backend/.env

# Rebuild backend
cd backend
npm run build

# Restart service (choose one)
pm2 restart backend              # If using PM2
docker-compose restart backend   # If using Docker
sudo systemctl restart business-talk-backend  # If using systemd

# Test
curl http://localhost:5000/api/mongodb/clusters

TROUBLESHOOTING
================================================================================

# If credentials not found:

# If backend not responding:

npm run build
pm2 start dist/index.js --name backend

# Check logs for errors:
pm2 logs backend
